📄 SmartIrrigation_Pro.ino
/*
 * Smart Irrigation System - Pro Version
 * مع جدولة، تحكم Serial، حماية متقدمة
 */

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>
#include <RTClib.h>

#define SOIL_PIN     A0
#define RAIN_PIN     2
#define DHT_PIN      7
#define RELAY_PIN    3
#define LED_PUMP     5
#define LED_RAIN     6
#define LED_ALERT    8
#define BUZZER_PIN   9

int moistureLow  = 40;
int moistureHigh = 75;
int wateringCycles = 0;
float totalWaterUsed = 0;

LiquidCrystal_I2C lcd(0x27, 16, 2);
DHT dht(DHT_PIN, DHT22);
RTC_DS3231 rtc;

// جداول الري (Schedule)
struct Schedule {
  int hour;
  int minute;
  int duration;  // بالدقائق
  bool enabled;
};

Schedule schedules[5] = {
  {6, 0, 5, true},   // 6:00 صباحاً - 5 دقائق
  {18, 30, 5, true},  // 6:30 مساءً - 5 دقائق
  {0, 0, 0, false},
  {0, 0, 0, false},
  {0, 0, 0, false}
};

bool pumpRunning = false;
unsigned long pumpStartTime = 0;

void setup() {
  Serial.begin(9600);
  
  pinMode(RAIN_PIN, INPUT);
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(LED_PUMP, OUTPUT);
  pinMode(LED_RAIN, OUTPUT);
  pinMode(LED_ALERT, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  
  digitalWrite(RELAY_PIN, HIGH);
  
  dht.begin();
  lcd.init();
  lcd.backlight();
  rtc.begin();
  
  Serial.println("=== Smart Irrigation Pro Started ===");
  printHelp();
}

void loop() {
  readSensors();
  checkSchedule();
  controlIrrigation();
  handleSerialCommands();
  updateDisplay();
  delay(1000);
}

void checkSchedule() {
  DateTime now = rtc.now();
  
  for (int i = 0; i < 5; i++) {
    if (schedules[i].enabled && 
        schedules[i].hour == now.hour() && 
        schedules[i].minute == now.minute() &&
        now.second() < 2) {
      
      Serial.print("[SCHEDULE] Triggered at ");
      Serial.print(now.hour());
      Serial.print(":");
      Serial.println(now.minute());
      
      runScheduledWatering(schedules[i].duration);
    }
  }
}

void runScheduledWatering(int minutes) {
  startPump();
  delay(minutes * 60 * 1000UL);
  stopPump("Schedule completed");
}

void handleSerialCommands() {
  if (!Serial.available()) return;
  
  String cmd = Serial.readStringUntil('\n');
  cmd.trim();
  cmd.toUpperCase();
  
  if (cmd == "PUMP_ON") startPump();
  else if (cmd == "PUMP_OFF") stopPump("Manual");
  else if (cmd == "STATUS") printStatus();
  else if (cmd == "RESET") resetSystem();
  else if (cmd.startsWith("SET_THRESHOLD")) {
    moistureLow = cmd.substring(14).toInt();
    Serial.print("Threshold set to: ");
    Serial.println(moistureLow);
  }
  else if (cmd == "HELP") printHelp();
  else Serial.println("Unknown command. Type HELP.");
}

void printStatus() {
  Serial.println("--- System Status ---");
  Serial.print("Moisture: "); Serial.println(readMoisture());
  Serial.print("Pump: "); Serial.println(pumpRunning ? "ON" : "OFF");
  Serial.print("Cycles: "); Serial.println(wateringCycles);
  Serial.print("Water used: "); Serial.println(totalWaterUsed);
}

void printHelp() {
  Serial.println("Commands:");
  Serial.println("  PUMP_ON, PUMP_OFF");
  Serial.println("  STATUS, RESET, HELP");
  Serial.println("  SET_THRESHOLD <value>");
}

/*
 * Smart Irrigation - ESP32 IoT Version
 * مع WiFi, MQTT, ThingSpeak, Telegram
 */

#include <WiFi.h>
#include <PubSubClient.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <DHT.h>

// إعدادات WiFi
const char* WIFI_SSID = "YourWiFiName";
const char* WIFI_PASS = "YourPassword";

// MQTT Broker (HiveMQ)
const char* MQTT_SERVER = "broker.hivemq.com";
const int   MQTT_PORT = 1883;
const char* MQTT_CLIENT = "smart_farm_001";

// Telegram Bot
const String BOT_TOKEN = "YOUR_BOT_TOKEN";
const String CHAT_ID = "YOUR_CHAT_ID";

// ThingSpeak
const String TS_API_KEY = "YOUR_API_KEY";

// Pin definitions
#define SOIL_PIN    34
#define RAIN_PIN    35
#define DHT_PIN     4
#define RELAY_PIN   5

WiFiClient espClient;
PubSubClient mqtt(espClient);
DHT dht(DHT_PIN, DHT22);

unsigned long lastUpload = 0;
const unsigned long UPLOAD_INTERVAL = 60000; // كل دقيقة

void setup() {
  Serial.begin(115200);
  
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, HIGH);
  
  dht.begin();
  
  setupWiFi();
  setupMQTT();
  sendTelegram("🌱 نظام الري بدأ التشغيل!");
}

void loop() {
  if (!mqtt.connected()) reconnectMQTT();
  mqtt.loop();
  
  // رفع البيانات كل دقيقة
  if (millis() - lastUpload > UPLOAD_INTERVAL) {
    readAndUpload();
    lastUpload = millis();
  }
  
  delay(100);
}

void setupWiFi() {
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  Serial.print("Connecting to WiFi");
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println();
  Serial.print("Connected! IP: ");
  Serial.println(WiFi.localIP());
}

void setupMQTT() {
  mqtt.setServer(MQTT_SERVER, MQTT_PORT);
  mqtt.setCallback(mqttCallback);
  reconnectMQTT();
}

void reconnectMQTT() {
  while (!mqtt.connected()) {
    if (mqtt.connect(MQTT_CLIENT)) {
      Serial.println("MQTT Connected");
      mqtt.subscribe("smart_farm/control/pump");
      mqtt.subscribe("smart_farm/control/threshold");
    } else {
      delay(5000);
    }
  }
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  
  String topicStr = String(topic);
  
  if (topicStr == "smart_farm/control/pump") {
    if (message == "ON") {
      digitalWrite(RELAY_PIN, LOW);
      sendTelegram("💧 المضخة شغّالة (تحكم عن بعد)");
    } else {
      digitalWrite(RELAY_PIN, HIGH);
      sendTelegram("⏹ المضخة متوقفة");
    }
  }
}

void readAndUpload() {
  int moisture = map(analogRead(SOIL_PIN), 4095, 0, 0, 100);
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();
  bool rain = (digitalRead(RAIN_PIN) == LOW);
  
  // نشر إلى MQTT
  mqtt.publish("smart_farm/sensors/moisture", String(moisture).c_str());
  mqtt.publish("smart_farm/sensors/temp", String(temp).c_str());
  mqtt.publish("smart_farm/sensors/rain", rain ? "YES" : "NO");
  
  // رفع لـ ThingSpeak
  uploadToThingSpeak(moisture, temp, hum, rain);
  
  // تنبيه إذا الرطوبة منخفضة جداً
  if (moisture < 20) {
    sendTelegram("⚠️ تحذير! رطوبة منخفضة جداً: " + String(moisture) + "%");
  }
}

void uploadToThingSpeak(int m, float t, float h, bool r) {
  HTTPClient http;
  String url = "https://api.thingspeak.com/update?api_key=" + TS_API_KEY;
  url += "&field1=" + String(m);
  url += "&field2=" + String(t);
  url += "&field3=" + String(h);
  url += "&field4=" + String(r ? 1 : 0);
  
  http.begin(url);
  int code = http.GET();
  http.end();
}

void sendTelegram(String message) {
  HTTPClient http;
  String url = "https://api.telegram.org/bot" + BOT_TOKEN;
  url += "/sendMessage?chat_id=" + CHAT_ID;
  url += "&text=" + message;
  
  http.begin(url);
  http.GET();
  http.end();
}
